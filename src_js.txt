
(function(){
"use strict";

// ==================== 数据层 ====================
var QUESTION_BANKS = {__BANKS__};

// ==================== 配置 ====================
var DIFF_NAMES = {1:'简单',2:'中等',3:'困难'};
var DIFF_ICONS = {1:'⭐',2:'📖',3:'🔥'};
var SUBJECT_KEYS = __KEYS__;
var COURSES = {__COURSES__};
var CHAPTER_NAMES = {__CH_NAMES__};
var currentCourse = '__COURSE__';

// ==================== 状态层 ====================
var currentSubject = '__SUBJECT__';
var ALL_QS = [];
var ALL_TERMS = [];
var STATS = {};
var QS = [], idx = 0, answers = {}, done = {}, viewMode = 'home';
var errSet = new Set();
var bookmarkSet = new Set();

// ==================== localStorage 辅助 ====================
function lsKey(purpose) {
  return currentSubject + '_' + purpose;
}
function lsGet(purpose) {
  try { return localStorage.getItem(lsKey(purpose)); } catch(e) { return null; }
}
function lsSet(purpose, val) {
  try { localStorage.setItem(lsKey(purpose), val); } catch(e) {}
}
function lsRemove(purpose) {
  try { localStorage.removeItem(lsKey(purpose)); } catch(e) {}
}
function lsGetJson(purpose) {
  try { var r = localStorage.getItem(lsKey(purpose)); return r ? JSON.parse(r) : null; } catch(e) { return null; }
}
function lsSetJson(purpose, obj) {
  try { localStorage.setItem(lsKey(purpose), JSON.stringify(obj)); } catch(e) {}
}

function saveErr() { lsSet('errors', JSON.stringify([...errSet])); }
function getErrCount() { return errSet.size; }
function clearErrSet() { errSet.clear(); saveErr(); }
function saveBookmarks() { lsSet('bookmarks', JSON.stringify([...bookmarkSet])); }
function getBookmarkCount() { return bookmarkSet.size; }
function getLastScore() { return lsGet('score'); }
function saveLastScore(s) { lsSet('score', s); }

// ==================== 科目切换 ====================
function loadSubjectData(key) {
  var bank = QUESTION_BANKS[key];
  ALL_QS = bank.questions.slice();
  ALL_TERMS = bank.terms.slice();
  STATS = bank.stats;
}

function switchCourse(courseKey) {
  if (courseKey === currentCourse) return;
  if (viewMode === 'quiz') saveProgress();
  saveErr(); saveBookmarks();
  currentCourse = courseKey;
  var firstCh = COURSES[courseKey].chapters[0];
  currentSubject = firstCh;
  loadSubjectData(firstCh);
  errSet = (function() { var r = lsGet('errors'); return r ? new Set(JSON.parse(r)) : new Set(); })();
  bookmarkSet = (function() { var r = lsGet('bookmarks'); return r ? new Set(JSON.parse(r)) : new Set(); })();
  QS = ALL_QS.slice(); idx = 0; answers = {}; done = {}; viewMode = 'home';
  buildCourseTabs();
buildChapterTabs();
    updateSubjectUI();
  
  showHome();
}

function switchChapter(chId) {
  // switchChapter
  if (chId === currentSubject && viewMode === 'home') return;
  if (viewMode === 'quiz') saveProgress();
  saveErr(); saveBookmarks();
  currentSubject = chId;
  loadSubjectData(chId);
  errSet = (function() { var r = lsGet('errors'); return r ? new Set(JSON.parse(r)) : new Set(); })();
  bookmarkSet = (function() { var r = lsGet('bookmarks'); return r ? new Set(JSON.parse(r)) : new Set(); })();
  QS = ALL_QS.slice(); idx = 0; answers = {}; done = {}; viewMode = 'home';
  updateSubjectUI();
  showHome();
}



function buildCourseBar() {{
  var h = '';
  Object.keys(COURSES).forEach(function(k) {{
    var c = COURSES[k];
    h += '<div class="course-btn' + (k === currentCourse ? ' active' : '') + '" data-course="' + k + '">' + c.name + '</div>';
  }});
  var el = document.getElementById('courseBar');
  if(el) el.innerHTML = h;
  // Delegate click
  if(el) el.addEventListener('click', function(e){{
    var btn = e.target.closest('.course-btn');
    if(btn) switchCourse(btn.getAttribute('data-course'));
  }});
}}


function buildCourseTabs() {
  var h = '';
  Object.keys(COURSES).forEach(function(k) {
    var course = COURSES[k];
    h += '<button class="subject-tab' + (k === currentCourse ? ' active' : '') + '" data-course="' + k + '">' + course.code + ' ' + course.name + '</button>';
  });
  var ss = document.getElementById('subjectSwitcher'); if(ss) ss.innerHTML = h;
  document.querySelectorAll('.subject-tab').forEach(function(t) {
    t.addEventListener('click', function() {
      switchCourse(t.getAttribute('data-course'));
    });
  });
}
function buildChapterTabs() {
  var chapters = COURSES[currentCourse].chapters;
  var h = '';
  chapters.forEach(function(chId) {
    var name = CHAPTER_NAMES[chId] || chId;
    var cls = 'chapter-tab' + (chId === currentSubject ? ' active' : '');
    h += '<button class="' + cls + '" data-ch="' + chId + '">' + name + '</button>';
  });
  var el = document.getElementById('chapterSwitcher');
  if (el) el.innerHTML = h;
  document.querySelectorAll('.chapter-tab').forEach(function(t) {
    t.addEventListener('click', function() {
      switchChapter(t.getAttribute('data-ch'));
    });
  });
}

function updateSubjectUI() {
  var course = COURSES[currentCourse];
  var bank = QUESTION_BANKS[currentSubject];
  var chName = CHAPTER_NAMES[currentSubject] || bank.chapterLabel || '';
  var bc = document.getElementById('brandCode'); if(bc) bc.innerHTML = '<em>' + course.code + '</em> ' + course.name;
  document.title = '湖南大学 ' + course.name + ' 考研题库';
  var tabs = document.querySelectorAll('.subject-tab');
  tabs.forEach(function(t) {
    t.classList.toggle('active', t.getAttribute('data-course') === currentCourse);
  });
  tabs = document.querySelectorAll('.chapter-tab');
  tabs.forEach(function(t) {
    t.classList.toggle('active', t.getAttribute('data-ch') === currentSubject);
  });
}

function saveProgress() {
  if (viewMode !== 'quiz' && viewMode !== 'diff') return;
  var doneIds = Object.keys(done); if (!doneIds.length) return;
  var mode = 'all';
  if (viewMode === 'diff' && QS.length && QS[0]) mode = 'diff:' + QS[0].difficulty;
  else if (QS.length !== ALL_QS.length) {
    var allErr = QS.every(function(q) { return errSet.has(q.id); });
    if (allErr) mode = 'err';
    else {
      var allChoice = QS.every(function(q) { return q.type === 'choice'; });
      if (allChoice) mode = 'choice';
      else { var allTF = QS.every(function(q) { return q.type === 'truefalse'; }); if (allTF) mode = 'truefalse'; }
    }
  }
  var p = { mode: mode, idx: idx, answers: answers, done: doneIds };
  lsSetJson('progress', p);
}

function loadProgress() {
  return lsGetJson('progress');
}

function clearProgress() {
  lsRemove('progress');
}

function resumeProgress() {
  var p = loadProgress(); if (!p) return false;
  answers = p.answers || {}; done = {};
  (p.done || []).forEach(function(id) { done[id] = true; });
  var mode = p.mode || 'all';
  if (mode === 'all') { QS = ALL_QS.slice(); viewMode = 'quiz'; }
  else if (mode === 'err') { QS = ALL_QS.filter(function(q) { return errSet.has(q.id); }); viewMode = 'quiz'; }
  else if (mode === 'choice') { QS = ALL_QS.filter(function(q) { return q.type === 'choice'; }); viewMode = 'quiz'; }
  else if (mode === 'truefalse') { QS = ALL_QS.filter(function(q) { return q.type === 'truefalse'; }); viewMode = 'quiz'; }
  else if (mode.indexOf('diff:') === 0) { var d = parseInt(mode.split(':')[1]); QS = ALL_QS.filter(function(q) { return q.difficulty === d; }); viewMode = 'diff'; }
  else { QS = ALL_QS.slice(); viewMode = 'quiz'; }
  if (!QS.length) { showHome(); return false; }
  idx = p.idx || 0; if (idx >= QS.length) idx = QS.length - 1;
  showQuizUI(); prog(); render(idx);
  return true;
}

// ==================== 辅助函数 ====================

// ==================== 自动错题收集 ====================
function autoCollectError(qid, userAnswer, correctAnswer) {
  var isCorrect = (userAnswer === correctAnswer);
  if (!isCorrect) {
    if (!errSet.has(qid)) { errSet.add(qid); saveErr(); }
  } else {
    if (errSet.has(qid)) { errSet.delete(qid); saveErr(); }
  }
}

// ==================== UI 控制 ====================
function showQuizUI() {
  document.getElementById('progWrap').style.display = 'flex';
  document.getElementById('navRow').style.display = 'flex';
}
function hideQuizUI() {
  document.getElementById('progWrap').style.display = 'none';
  document.getElementById('navRow').style.display = 'none';
}


// ==================== 渲染 ====================
var quizEl, progWrap, progBar, progText, progDiff, progModeTag, navRow, navHint, btnPrev, btnNext, btnHome;

function prog() {
  var n = Object.keys(done).length, t = QS.length;
  progBar.style.width = Math.round(n / t * 100) + '%';
  progText.textContent = n + ' / ' + t;
  if (viewMode === 'diff' && QS[0]) {
    progDiff.style.display = ''; progDiff.textContent = '🎯 ' + DIFF_NAMES[QS[0].difficulty];
  } else progDiff.style.display = 'none';
  progModeTag.style.display = 'none';
}

function nav() {
  btnPrev.disabled = (idx === 0);
  var q = QS[idx]; if (!q) return;
  var ad = q.id in done;
  if (!ad) {
    btnNext.disabled = true;
    navHint.textContent = '🔒 请先选择答案';
  } else if (idx >= QS.length - 1 && QS.every(function(q) { return q.id in done; })) {
    btnNext.disabled = false;
    navHint.textContent = '全部完成！';
  } else {
    btnNext.disabled = false;
    btnNext.textContent = '下一题 ▶';
    navHint.textContent = '';
  }
}

function render(i) {
  try {
  if (i < 0 || i >= QS.length) { showResult(); return; }
  idx = i; showQuizUI();
  var q = QS[i], sel = answers[q.id] || null, ad = q.id in done;
  var dName = DIFF_NAMES[q.difficulty], dCls = q.difficulty === 1 ? 'easy' : (q.difficulty === 3 ? 'hard' : 'mid');
  var isTF = (q.type === 'truefalse');
  var typeLabel = isTF ? '判断题' : '选择题';
  var typeCls = isTF ? 'tf' : 'choice';
  var correctAnswer = q.answer;

  var h = '<div class="card"><div class="q-head">';
  h += '<span class="q-num">第 ' + q.id + ' 题</span>';
  h += '<span class="q-type ' + typeCls + '">' + typeLabel + '</span>';
  h += '<span class="q-diff ' + dCls + '">' + DIFF_ICONS[q.difficulty] + ' ' + dName + '</span>';
  h += '<button class="err-toggle' + (bookmarkSet.has(q.id) ? ' on' : '') + '" id="errBtn">' + (bookmarkSet.has(q.id) ? '⭐ 已标记' : '☆ 标记重点') + '</button>';
  h += '</div>';
  h += '<div class="q-text">' + q.question + '</div>';
  // Tags
  if (q.tags && q.tags.length) {
    h += '<div class="q-tags">';
    q.tags.forEach(function(tag) { h += '<span class="q-tag">' + tag + '</span>'; });
    h += '</div>';
  }
  if (isTF) {
    var tCls = 'tf-btn t', fCls = 'tf-btn f';
    if (ad) {
      tCls += ' done'; fCls += ' done';
      if (correctAnswer === true) tCls += ' ok'; else fCls += ' ok';
      if (sel === true && correctAnswer !== true) tCls += ' bad';
      if (sel === false && correctAnswer !== false) fCls += ' bad';
    } else if (sel === true) tCls += ' sel-t'; else if (sel === false) fCls += ' sel-f';
    h += '<div class="tf-row"><div class="' + tCls + '" data-v="true">✅ 正确</div><div class="' + fCls + '" data-v="false">❌ 错误</div></div>';
  } else {
    var order = Object.keys(q.options);
    h += '<div class="options">';
    order.forEach(function(L) {
      var cls = 'opt';
      if (ad) {
        cls += ' done';
        if (L === correctAnswer) cls += ' ok';
        else if (L === sel && L !== correctAnswer) cls += ' bad';
      } else if (L === sel) cls += ' sel';
      h += '<div class="' + cls + '" data-l="' + L + '"><span class="opt-letter">' + L + '</span><span class="opt-label">' + q.options[L] + '</span></div>';
    });
    h += '</div>';
  }

  if (ad) h += '<div class="explain show" id="exp' + q.id + '"><strong>📖 解析：</strong>' + q.explanation + '</div>';
  else h += '<div class="explain" id="exp' + q.id + '"></div>';
  if (!ad) h += '<div class="lock-hint">🔒 选择答案后才能前进</div>';

  quizEl.innerHTML = h;

  // 事件绑定
  var eb = document.getElementById('errBtn'); if (eb) eb.addEventListener('click', function(e) { e.stopPropagation(); toggleBookmark(q.id); });
  if (isTF) {
    quizEl.querySelectorAll('.tf-btn').forEach(function(o) {
      o.addEventListener('click', function() {
        pickTF(q.id, o.getAttribute('data-v') === 'true');
      });
    });
  } else {
    quizEl.querySelectorAll('.opt').forEach(function(o) {
      o.addEventListener('click', function() {
        pick(q.id, o.getAttribute('data-l'));
      });
    });
  }
  nav(); prog();
    } catch(e) { document.getElementById('quiz').innerHTML = '<h2 style="color:red;padding:20px">Render Error: '+e.message+'</h2>'; }
}

function toggleBookmark(qid) {
  bookmarkSet.has(qid) ? bookmarkSet.delete(qid) : bookmarkSet.add(qid);
  saveBookmarks();
  var et = document.getElementById('errBtn'); if (et) {
    et.textContent = bookmarkSet.has(qid) ? '⭐ 已标记' : '☆ 标记重点';
    et.className = 'err-toggle' + (bookmarkSet.has(qid) ? ' on' : '');
  }
}

// ==================== 答案处理 ====================
function pick(qid, L) { _answer(qid, L); }
function pickTF(qid, v) { _answer(qid, v); }

function _answer(qid, val) {
  // 考试模式不允许修改答案

  var prevAnswer = answers[qid];
  answers[qid] = val;
  done[qid] = true;

  var q = QS.find(function(x) { return x.id === qid; });
  if (!q) return;

  // 自动错题收集
  autoCollectError(qid, val, q.answer);

  saveProgress();

  render(idx);
  prog(); nav();
  // 修改答案后更新导航状态
  if (prevAnswer !== undefined) {
    navHint.textContent = '✅ 答案已更新';
  }
}


// ==================== 导航 ====================
function prev() { if (idx > 0) render(idx - 1); }
function next() {
  var q = QS[idx]; if (!q) return;
  if (!(q.id in done)) return;
  if (idx >= QS.length - 1 && QS.every(function(q) { return q.id in done; })) showResult();
  else if (idx < QS.length - 1) render(idx + 1);
}

// ==================== 考试成绩 ====================
function showResult() {
  if (QS.length === 0) { alert('当前没有题目，请重新选择'); showHome(); return; }
  hideQuizUI(); viewMode = 'result';
  var cor = 0, wro = 0, una = 0;
  QS.forEach(function(q) {
    var ua = answers[q.id];
    if (ua === undefined) { una++; }
    else if (ua === q.answer) { cor++; }
    else { wro++; }
  });
  var sc = QS.length > 0 ? Math.round(cor / QS.length * 100) : 0;
  saveHistory(sc, cor, wro, una);
  saveLastScore(String(sc));
  clearProgress();
  var gd, gc;
  if (sc >= 90) { gd = '🏅 太棒了！掌握得非常扎实！'; gc = '#2D8C4A' }
  else if (sc >= 80) { gd = '👍 表现优秀！继续巩固薄弱知识点。'; gc = '#8B1A2B' }
  else if (sc >= 70) { gd = '📚 良好水平，建议针对错题强化复习。'; gc = '#C4924A' }
  else if (sc >= 60) { gd = '📖 刚刚及格，需要系统回顾本章内容。'; gc = '#E67E22' }
  else { gd = '💪 还需努力，建议重新梳理知识点后再做一遍。'; gc = '#D43D3D' }

  var bank = QUESTION_BANKS[currentSubject];
  var h = '<div class="result-card">';
  h += '<h2>📊 答题成绩</h2>';
  h += '<p style="font-size:.78rem;color:var(--grey);margin-bottom:8px">' + bank.name + ' · ' + bank.chapterLabel + '</p>';
  h += '<div class="score-circle" style="background:linear-gradient(135deg,' + gc + ',' + gc + 'dd)">' + sc + '<small>分</small></div>';
  h += '<p class="grade-text" style="color:' + gc + '">' + gd + '</p>';
  h += '<div class="stats"><span style="color:#2D8C4A">✅ 正确：<b>' + cor + '</b></span><span style="color:#D43D3D">❌ 错误：<b>' + wro + '</b></span>';
  if (una > 0) h += '<span style="color:var(--grey)">⬜ 未答：<b>' + una + '</b></span>';
  h += '</div>';
  h += '<p style="margin-top:6px;font-size:.82rem;color:var(--grey)">📌 错题已自动收录 <b style="color:var(--red)">' + getErrCount() + '</b> 题 | ⭐ 标记重点 <b style="color:var(--gold)">' + getBookmarkCount() + '</b> 题</p>';

  // 逐题回顾（可折叠）
  h += '<div class="review-list"><h3>📋 逐题回顾（点击展开）</h3>';
  QS.forEach(function(q) {
    var ua = answers[q.id], ok = ua === q.answer;
    var icon = ua !== undefined ? (ok ? '✅' : '❌') : '⬜';
    var ansLabel = q.type === 'truefalse' ? (q.answer ? '✅正确' : '❌错误') : q.answer;
    var uaL = q.type === 'truefalse' ? (ua === true ? '正确' : (ua === false ? '错误' : '')) : ua;
    h += '<details class="review-detail"><summary class="review-item">' + icon + ' <b>第' + q.id + '题</b> ';
    if (ua !== undefined && !ok) h += '<span style="color:#D43D3D">（你：' + uaL + '，正确：' + ansLabel + '）</span>';
    else if (ua !== undefined && ok) h += '<span style="color:#2D8C4A">（' + uaL + '）</span>';
    h += '</summary><div class="review-body">';
    h += '<p><b>题目：</b>' + q.question + '</p>';
    if (q.type === 'choice') {
      Object.keys(q.options).forEach(function(L) {
        h += '<p>' + (L === q.answer ? '✅' : '  ') + ' ' + L + '. ' + q.options[L] + '</p>';
      });
    } else {
      h += '<p>正确答案：' + (q.answer ? '✅ 正确' : '❌ 错误') + '</p>';

    }
    h += '<p><b>解析：</b>' + q.explanation + '</p>';
    h += '</div></details>';
  });
  h += '</div>';

  h += '<div class="result-actions">';
  h += '<button class="btn btn-go" id="btnRetry">🔄 重新做题</button>';
  if (wro > 0) h += '<button class="btn btn-outline-danger" id="btnWrong">📝 只做错题</button>';
  h += '<button class="btn btn-ghost" id="btnHome2">🏠 返回首页</button>';
  h += '</div></div>';
  quizEl.innerHTML = h;
  document.getElementById('btnRetry').addEventListener('click', retrySame);
  var bw = document.getElementById('btnWrong'); if (bw) bw.addEventListener('click', doWrongOnly);
  document.getElementById('btnHome2').addEventListener('click', showHome);

}

function retrySame() { idx = 0; answers = {}; done = {}; quizEl.innerHTML = ''; showQuizUI(); prog(); render(0); }
function doWrongOnly() {
  var wl = QS.filter(function(q) { return (q.id in done) && answers[q.id] !== q.answer; });
  if (!wl.length) { alert('没有错题！🎉'); return; }
  QS = wl; viewMode = 'quiz'; idx = 0; answers = {}; done = {}; showQuizUI(); prog(); render(0);
}

// ==================== 历史记录 ====================
function saveHistory(sc, cor, wro, una) {
  var records = lsGetJson('history') || [];
  var timeUsed = 0;
  records.unshift({
    id: Date.now(),
    date: new Date().toISOString(),
    subject: currentSubject,
    subjectName: QUESTION_BANKS[currentSubject].name,
    mode: 'practice',
    total: QS.length,
    correct: cor,
    wrong: wro,
    unanswered: una,
    score: sc,
    timeUsed: timeUsed
  });
  // 只保留最近50条
  if (records.length > 50) records = records.slice(0, 50);
  lsSetJson('history', records);
}

function showHistory() {
  hideQuizUI(); viewMode = 'history';
  var records = lsGetJson('history') || [];
  var h = '<div class="home-card"><h2>📋 历史记录</h2>';
  h += '<div class="filter-row">';
  h += '<button class="btn btn-sm btn-ghost hist-filter active" data-subj="all">全部</button>';
  SUBJECT_KEYS.forEach(function(k) {
    h += '<button class="btn btn-sm btn-ghost hist-filter" data-subj="' + k + '">' + QUESTION_BANKS[k].name + '</button>';
  });
  h += '</div>';
  if (!records.length) {
    h += '<p style="color:var(--grey);margin:30px 0">暂无历史记录</p>';
  } else {
    h += '<div style="overflow-x:auto"><table class="history-table"><thead><tr><th>日期</th><th>科目</th><th>模式</th><th>题数</th><th>正确</th><th>错误</th><th>正确率</th></tr></thead><tbody id="histBody">';
    records.forEach(function(r) {
      var modeLabel = '📖练习';
      h += '<tr class="hist-row" data-subj="' + r.subject + '" data-id="' + r.id + '"><td>' + r.date.slice(0,16).replace('T',' ') + '</td><td>' + r.subjectName + '</td><td>' + modeLabel + '</td><td>' + r.total + '</td><td style="color:#2D8C4A">' + r.correct + '</td><td style="color:#D43D3D">' + r.wrong + '</td><td><b style="color:' + (r.score >= 80 ? '#2D8C4A' : r.score >= 60 ? '#C4924A' : '#D43D3D') + '">' + r.score + '%</b></td></tr>';
    });
    h += '</tbody></table></div>';
  }
  h += '<div style="text-align:center;margin-top:20px"><button class="btn btn-ghost" id="btnHistBack">← 返回首页</button>';
  if (records.length > 0) h += ' <button class="btn btn-outline-danger btn-sm" id="btnClearHist">清空历史</button>';
  h += '</div></div>';
  quizEl.innerHTML = h;
  document.getElementById('btnHistBack').addEventListener('click', showHome);
  var cb = document.getElementById('btnClearHist'); if (cb) cb.addEventListener('click', function() { if (confirm('确定清空所有历史记录？')) { lsRemove('history'); showHistory(); } });
  // 科目筛选
  document.querySelectorAll('.hist-filter').forEach(function(btn) {
    btn.addEventListener('click', function() {
      document.querySelectorAll('.hist-filter').forEach(function(b) { b.classList.remove('active'); });
      btn.classList.add('active');
      var subj = btn.getAttribute('data-subj');
      document.querySelectorAll('.hist-row').forEach(function(row) {
        row.style.display = (subj === 'all' || row.getAttribute('data-subj') === subj) ? '' : 'none';
      });
    });
  });

function checkResumeQuiz(callback) {
  var prog = loadProgress();

function migrateOldData() {
  var migrations = [
    { old: 'biochem_err_v1', newSubj: 'biochemistry', purpose: 'errors' },
    { old: 'biochem_score_v1', newSubj: 'biochemistry', purpose: 'score' },
    { old: 'cellbio_err_v2', newSubj: 'cellbiology', purpose: 'errors' },
    { old: 'cellbio_score_v2', newSubj: 'cellbiology', purpose: 'score' },
  ];
  migrations.forEach(function(m) {
    try {
      var oldVal = localStorage.getItem(m.old);
      if (oldVal !== null) {
        var newKey = m.newSubj + '_' + m.purpose;
        if (!localStorage.getItem(newKey)) {
          localStorage.setItem(newKey, oldVal);
        }
        localStorage.removeItem(m.old);
      }
    } catch(e) {}
  });
  // 清理旧进度数据
  try { localStorage.removeItem('biochem_progress_v1'); } catch(e) {}
  try { localStorage.removeItem('cellbio_progress_v1'); } catch(e) {}
}

function generateIcon() {
  try {
    var size = 192;
    var canvas = document.createElement('canvas');
    canvas.width = size; canvas.height = size;
    var ctx = canvas.getContext('2d');
    // 红色背景
    ctx.fillStyle = '#8B1A2B';
    ctx.beginPath(); ctx.arc(size/2, size/2, size/2, 0, Math.PI*2); ctx.fill();
    // 白色文字
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 72px "PingFang SC","Microsoft YaHei",sans-serif';
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText('湖大', size/2, size/2);
    return canvas.toDataURL('image/png');
  } catch(e) { return null; }
}

// 3. Service Worker (Cache-First)
function registerSW() {
  if (!('serviceWorker' in navigator)) return;
  try {
    var CACHE_NAME = 'huda-quiz-v1';
    var swCode = [
      'var CACHE_NAME = "' + CACHE_NAME + '";',
      'self.addEventListener("install", function(e) {',
      '  e.waitUntil(self.skipWaiting());',
      '});',
      'self.addEventListener("activate", function(e) {',
      '  e.waitUntil(caches.keys().then(function(keys) {',
      '    return Promise.all(keys.filter(function(k) { return k !== CACHE_NAME; }).map(function(k) { return caches.delete(k); }));',
      '  }).then(function() { return self.clients.claim(); }));',
      '});',
      'self.addEventListener("fetch", function(e) {',
      '  if (e.request.method !== "GET") return;',
      '  e.respondWith(',
      '    caches.match(e.request).then(function(cached) {',
      '      if (cached) return cached;',
      '      return fetch(e.request).then(function(response) {',
      '        if (response && response.status === 200 && response.type === "basic") {',
      '          var clone = response.clone();',
      '          caches.open(CACHE_NAME).then(function(cache) { cache.put(e.request, clone); });',
      '        }',
      '        return response;',
      '      }).catch(function() {',
      '        return new Response("离线模式 - 资源未缓存", {status:503});',
      '      });',
      '    })',
      '  );',
      '});'
    ].join('\n');
    var swBlob = new Blob([swCode], {type: 'application/javascript'});
    var swUrl = URL.createObjectURL(swBlob);
    navigator.serviceWorker.register(swUrl).then(function(reg) {
      // 注册成功后立即缓存当前页面
      if (reg.installing) {
        reg.installing.addEventListener('statechange', function() {
          if (this.state === 'activated') cacheCurrentPage();
        });
      } else if (reg.active) {
        cacheCurrentPage();
      }
    }).catch(function(e) {
      // 静默降级
    });
  } catch(e) {}
}

function cacheCurrentPage() {
  try {
    if ('caches' in window) {
      caches.open('huda-quiz-v1').then(function(cache) {
        cache.add(window.location.href).catch(function() {});
      });
    }
  } catch(e) {}
}
