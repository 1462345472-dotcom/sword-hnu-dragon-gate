/* ===== 统计 ===== */
function chStats(key){
  var b=getBank(key);if(!b)return{t:0,a:0,c:0};
  var qs=b.questions,t=qs.length,a=0,c=0;
  for(var i=0;i<t;i++){
    if(S.answers[qs[i].id]!==undefined)a++;
    var sq=qs[i];var sa=S.answers[sq.id];
    if(sq.type==='truefalse'){if(String(sa).toLowerCase()===String(sq.answer).toLowerCase())c++;}
    else if(sq.type==='short'){if(sa==='done')c++;}
    else{if(sa===sq.answer)c++;}
  }return{t:t,a:a,c:c};
}
function courseStats(ck){

  for(var i=0;i<chs.length;i++){var s=chStats(chs[i]);t+=s.t;a+=s.a;c+=s.c;}
  return{t:t,a:a,c:c};
}

function shuffle(a){return a.slice();}
function startQuiz(key,mode){
  S.subject=key;S.quizMode=mode||'all';S.qIndex=0;S.answers={};S.revealed={};S.streak=0;
  if(mode==='wrong')S.questions=shuffle(wrongQs());
  else if(mode==='bookmarked')S.questions=shuffle(bmQs());
  else if(mode==='choice'||mode==='truefalse'||mode==='multi'||mode==='short'){
    var qs=allQs();var ft=mode;
    S.questions=shuffle(qs.filter(function(q){return q.type===ft;}));
  }
  else S.questions=shuffle(allQs());
  if(S.questions.length===0){toast('没有可刷的题目');return false;}
  return true;
}
/* ===== 续练系统 ===== */
function progressKey(subjectKey,mode){return subjectKey+'|'+(mode||'all');}

/* 带进度检查的刷题入口 */
function startQuizWithProgress(subjectKey,mode){
  S._pendingSubject=subjectKey;S._pendingMode=mode||'all';
  var pk=progressKey(subjectKey,mode||'all');
  var saved=S.savedProgress[pk];
  if(saved&&saved.answers&&Object.keys(saved.answers).length>0){
    /* 有保存的进度 - 弹窗让用户选择继续还是从头 */
    showResumePrompt(saved);
  }else{
    startFreshQuiz();
  }
}

/* 显示续练提示弹窗 */
function showResumePrompt(saved){
  var answered=Object.keys(saved.answers).length;
  var qIndex=saved.qIndex||0;
  /* 用confirm简单实现，也可改为自定义弹窗 */
  var msg='检测到上次练习进度：\n已完成 '+answered+' 题，上次做到第 '+(qIndex+1)+' 题。\n\n点击"确定"继续上次，点击"取消"从头开始。';
  if(confirm(msg)){
    resumeSavedProgress();
  }else{
    clearSavedProgress();
    startFreshQuiz();
  }
}

/* 恢复保存的进度 */
function resumeSavedProgress(){
  var subjectKey=S._pendingSubject;var mode=S._pendingMode;
  var pk=progressKey(subjectKey,mode);
  var saved=S.savedProgress[pk];
  S.subject=subjectKey;S.quizMode=mode;S._multiSelection={};
  if(saved&&saved.answers){
    S.answers=saved.answers||{};
    S.revealed=saved.revealed||{};
    S.qIndex=saved.qIndex||0;
    S.streak=saved.streak||0;
    /* 重建题目列表 - 按模式过滤 */
    var bank=getBank(subjectKey);var src=bank?bank.questions:[];
    if(mode==='wrong')src=wrongQs();
    else if(mode==='bookmarked')src=bmQs();
    else if(mode==='choice')src=src.filter(function(q){return q.type==='choice';});
    else if(mode==='truefalse')src=src.filter(function(q){return q.type==='truefalse';});
    else if(mode==='multi')src=src.filter(function(q){return q.type==='multi';});
    else if(mode==='short')src=src.filter(function(q){return q.type==='short';});
    S.questions=shuffle(src);
    if(S.questions.length===0){toast('没有可刷的题目');return;}
    if(S.qIndex>=S.questions.length)S.qIndex=0;
    /* 清理不在此次题库的陈旧答案 */
    var curIds={};for(var i=0;i<S.questions.length;i++){curIds[S.questions[i].id]=true;}
    var cleanA={},cleanR={};
    for(var id in S.answers){if(curIds[id]){cleanA[id]=S.answers[id];if(S.revealed[id])cleanR[id]=true;}}
    S.answers=cleanA;S.revealed=cleanR;
  }else{
    startFreshQuiz();
    return;
  }
  switchView('quiz');
}

/* 清除当前章节+模式的进度 */
function clearSavedProgress(){
  var pk=progressKey(S._pendingSubject,S._pendingMode);
  delete S.savedProgress[pk];saveState();
}

/* 全新开始 */
function startFreshQuiz(){
  S._multiSelection={};
  var subjectKey=S._pendingSubject;var mode=S._pendingMode;
  var pk=progressKey(subjectKey,mode);
  delete S.savedProgress[pk];saveState();
  if(!startQuiz(subjectKey,mode))return;
  switchView('quiz');
}

/* 章节专项 (按题型过滤) */
function startChapterMode(subjectKey,filterType){
  var b=getBank(subjectKey);if(!b){toast('章节数据未找到');return;}
  var filtered=b.questions.filter(function(q){return q.type===filterType;});
  if(filtered.length===0){toast('该章节暂无此题型');return;}
  S._pendingSubject=subjectKey;S._pendingMode=filterType;
  var pk=progressKey(subjectKey,filterType);
  var saved=S.savedProgress[pk];
  if(saved&&saved.answers&&Object.keys(saved.answers).length>0){
    showResumePrompt(saved);
  }else{
    /* 用过滤后的题表启动 */
    S.subject=subjectKey;S.quizMode=filterType;S.qIndex=0;S.answers={};S.revealed={};S.streak=0;
    S.questions=shuffle(filtered);
    if(S.questions.length===0){toast('没有可刷的题目');return;}
    switchView('quiz');
  }
}

/* 保存当前进度到快照 */
function saveQuizProgress(){
  if(!S.subject||S.questions.length===0)return;
  var pk=progressKey(S.subject,S.quizMode);
  S.savedProgress[pk]={
    qIndex:S.qIndex,
    answers:S.answers,
    revealed:S.revealed,
    streak:S.streak,
    timestamp:new Date().toISOString()
  };
  saveState();
}

function curQ(){return S.qIndex<S.questions.length?S.questions[S.qIndex]:null;}
function isAnswered(qId){return S.answers[qId]!==undefined;}
function isRevealed(qId){return S.revealed[qId]===true;}

function submitAnswer(qId,ua){
  if(isAnswered(qId))return;
  S.answers[qId]=ua;S.revealed[qId]=true;
  var q=curQ();var correct=(ua===q.answer);
  if(correct){S.streak++;if(S.streak>S.bestStreak){S.bestStreak=S.streak;}spawnMapleParticles();if(S.streak%5===0)showMilestone(S.streak);checkAchievements();}
  else{S.streak=0;S.wrongSet[qId]=true;}
  /* 每次答题后保存进度 */
  saveQuizProgress();saveState();
  return correct;
}

var ACHIEVEMENTS = [

function checkAchievements(){
  var unlocked=false;
  for(var i=0;i<ACHIEVEMENTS.length;i++){
    var a=ACHIEVEMENTS[i];
    if(S.achievements[a.id])continue;
    var met=false;
    if(a.total){
      var t=0;var ks=Object.keys(QUESTION_BANKS);
      for(var j=0;j<ks.length;j++){
        var qs=QUESTION_BANKS[ks[j]].questions;
        for(var k=0;k<qs.length;k++){if(S.answers[qs[k].id]===qs[k].answer)t++;}
      }
      if(t>=a.threshold)met=true;
    }else{if(S.streak>=a.threshold)met=true;}
    if(met){S.achievements[a.id]=true;unlocked=true;showAchievement(a);}
  }
  if(unlocked)saveState();
}

}

  var answered=isAnswered(q.id);var revealed=isRevealed(q.id);
  var ua=S.answers[q.id];
  var isCorrect;
  if(q.type==='multi'){isCorrect=(ua===q.answer);}
  else if(q.type==='short'){isCorrect=(ua==='done');}
  else if(q.type==='truefalse'){isCorrect=(String(ua).toLowerCase()===String(q.answer).toLowerCase());}
  else{isCorrect=(ua===q.answer);}
  var isBM=S.bookmarks[q.id];

  /* ============================================================
     renderQuiz · 满屏重构
     frontend-design: 剑指湖大四色选项+色相锁定+反AI痕迹

          '<span class="q-tag q-tag-diff '+diffClass(q.difficulty)+'">'+diffLabel(q.difficulty)+'</span>'+

          '<button class="btn btn-secondary btn-sm" data-action="nav-prev" '+(idx===0?'disabled':'')+'>上一题</button>'+
          '<span class="spacer"></span>'+
          '<button class="btn btn-secondary btn-sm" data-action="nav-next">'+(idx>=total-1?'完成':'下一题')+'</button>'+
        '</div>'+
      '</div>'+
    '</div>';
}

function finishQuiz(){
  var qs=S.questions;var total=qs.length,correct=0;
  for(var i=0;i<total;i++){if(S.answers[qs[i].id]===qs[i].answer)correct++;}
  /* 完成一轮后清除该章节+模式的进度快照 */
  var pk=progressKey(S.subject,S.quizMode);
  delete S.savedProgress[pk];
  checkAchievements();saveState();
  return{total:total,score:correct,pct:Math.round(correct/total*100)};
}

/* ============================================================
   事件处理
   ============================================================ */
function handleClick(e){
  var t=e.target;var a=t.closest('[data-action]');if(!a)return;
  var act=a.getAttribute('data-action');

  switch(act){
    case'enter':switchView('home');break;
    case'switch-course':
      S.course=a.getAttribute('data-course');
      /* 切换科目时自动选该科目第一个章节 */
      var nc=COURSES[S.course];
      if(nc&&nc.chapters.length>0)S.subject=nc.chapters[0];
      saveState();renderHome();setTimeout(function(){var hw=document.getElementById('homeWrap');if(hw)hw.classList.add('animate-in');},50);
      break;
    case'select-chapter':
      /* 仅选择章节，不跳转 */
      S.subject=a.getAttribute('data-key');
      saveState();renderHome();setTimeout(function(){var hw=document.getElementById('homeWrap');if(hw)hw.classList.add('animate-in');},50);
      break;
    case'start-quiz':
      /* 从当前选中章节开始全部刷题 */
      startQuizWithProgress(S.subject,'all');
      break;
    case'start-choice':
      /* 当前章节的选择题专项 */
      startChapterMode(S.subject,'choice');
      break;
    case'start-truefalse':
      /* 当前章节的判断题专项 */
      startChapterMode(S.subject,'truefalse');
      break;
    case'start-multi':
      /* 当前章节的多选题专项 */
      startChapterMode(S.subject,'multi');
      break;
    case'start-short':
      /* 当前章节的简答题专项 */
      startChapterMode(S.subject,'short');
      break;
    case'start-wrong':
      var w=wrongQs();if(w.length===0){toast('暂无错题');return;}
      var wb=findBankForQ(w[0].id);
      if(wb)startQuizWithProgress(wb.key,'wrong');
      break;
    case'resume-quiz':
      /* 继续上次 - 从保存的进度恢复 */
      resumeSavedProgress();
      break;
    case'restart-quiz':
      /* 从头开始 - 清除进度重新来 */
      clearSavedProgress();
      startFreshQuiz();
      break;
    case'view-terms':S.termFilter='all';switchView('terms');break;
    case'answer':
      if(a.classList.contains('locked'))return;
      var v=a.getAttribute('data-value');var q=curQ();if(!q)return;
      var ua=v;if(q.type==='truefalse')ua=(String(v).toLowerCase()==='true');
      var correct=submitAnswer(q.id,ua);
      var optEl=a.closest('.option')||a.closest('.tf-btn');
      if(optEl)ripple(optEl,e);
      if(correct){
        var rect=optEl?optEl.getBoundingClientRect():null;
        if(rect)spawnParticles(rect.left+rect.width/2,rect.top+rect.height/2,18);
        else spawnParticles(window.innerWidth/2,window.innerHeight/2,18);
      }
      renderQuiz();
      setTimeout(function(){var ex=document.querySelector('.explanation');if(ex)ex.scrollIntoView({behavior:'smooth',block:'nearest'});},150);
      break;
    case'multi-toggle':
      if(a.classList.contains('locked'))return;
      var mv=a.getAttribute('data-value');var mq=curQ();if(!mq)return;
      if(!S._multiSelection)S._multiSelection={};
      if(!S._multiSelection[mq.id])S._multiSelection[mq.id]=[];
      var msel=S._multiSelection[mq.id];
      var mIdx=msel.indexOf(mv);
      if(mIdx>=0){msel.splice(mIdx,1);}else{msel.push(mv);}
      renderQuiz();
      break;
    case'multi-confirm':
      var mq2=curQ();if(!mq2)return;
      var sel=(S._multiSelection&&S._multiSelection[mq2.id])?S._multiSelection[mq2.id].slice():[];
      if(sel.length===0)return;
      sel.sort();var uaStr=sel.join('');
      submitAnswer(mq2.id,uaStr);
      delete S._multiSelection[mq2.id];
      renderQuiz();
      var mqEl=document.querySelector('.option.multi-option.correct');
      if(mqEl)setTimeout(function(){mqEl.scrollIntoView({behavior:'smooth',block:'nearest'});},150);
      break;
    case'short-reveal':
      var sq=curQ();if(!sq)return;
      submitAnswer(sq.id,'done');
      renderQuiz();
      setTimeout(function(){var ex=document.querySelector('.short-answer-reveal');if(ex)ex.scrollIntoView({behavior:'smooth',block:'nearest'});},150);
      break;
    case'nav-prev':
      if(S.qIndex>0){S.qIndex--;renderQuiz();slideCard('prev');$('view-quiz').scrollTop=0;}
      break;
    case'nav-next':
      if(S.qIndex<S.questions.length-1){S.qIndex++;renderQuiz();slideCard('next');$('view-quiz').scrollTop=0;}
      else{finishQuiz();switchView('result');}
      break;
    case'toggle-bookmark':
      var cq=curQ();if(!cq)return;
      if(S.bookmarks[cq.id])delete S.bookmarks[cq.id];else S.bookmarks[cq.id]=true;
      saveState();renderQuiz();
      break;
    case'show-sheet':showAnswerSheet();break;
    case'jump-to':
      var idx=parseInt(a.getAttribute('data-idx'));
      if(idx>=0&&idx<S.questions.length){S.qIndex=idx;renderQuiz();hideAnswerSheet();$('view-quiz').scrollTop=0;}
      break;
    case'go-home':switchView('home');break;
    case'filter-terms':S.termFilter=a.getAttribute('data-key');renderTerms();break;
    case'retry':if(startQuiz(S.subject,S.quizMode))switchView('quiz');break;
    case'review-wrong':if(startQuiz(S.subject,'wrong'))switchView('quiz');break;
    case'redo-wrong':
      var rqs=wrongQs();if(rqs.length===0){toast('暂无错题');return;}
      var rb=findBankForQ(rqs[0].id);
      if(rb&&startQuiz(rb.key,'wrong'))switchView('quiz');
      break;
    case'review-error':
      var rid=parseInt(a.getAttribute('data-id'));var rq=getQ(rid);
      if(!rq)return;var rrb=findBankForQ(rid);
      if(rrb){S.subject=rrb.key;S.questions=[rq];S.qIndex=0;S.answers={};S.revealed={};switchView('quiz');}
      break;
    case'clear-data':
      if(confirm('确定清除所有学习数据？此操作不可恢复。')){
        S.wrongSet={};S.bookmarks={};S.bestStreak=0;S.achievements={};
        saveState();toast('数据已清除');
        if(S.view==='home')renderHome();else if(S.view==='errors')renderErrors();else switchView('home');
        hideSettings();
      }
      break;
  }
}

function findBankForQ(qid){
  var ks=Object.keys(QUESTION_BANKS);
  for(var i=0;i<ks.length;i++){
    var bk=QUESTION_BANKS[ks[i]];
    for(var j=0;j<bk.questions.length;j++){if(bk.questions[j].id===qid)return bk;}
  }
  return null;
}

/* ===== 触摸滑动 ===== */
function handleTouchStart(e){
  if(S.view!=='quiz')return;
  S.touchStartX=e.touches[0].clientX;S.touchStartY=e.touches[0].clientY;S.touchMoved=false;
}
function handleTouchEnd(e){
  if(S.view!=='quiz'||S.touchMoved)return;
  var dx=e.changedTouches[0].clientX-S.touchStartX;
  var dy=e.changedTouches[0].clientY-S.touchStartY;
  if(Math.abs(dx)>Math.abs(dy)&&Math.abs(dx)>60){
    if(dx<0&&S.qIndex<S.questions.length-1){S.qIndex++;renderQuiz();slideCard('next');$('view-quiz').scrollTop=0;}
    else if(dx>0&&S.qIndex>0){S.qIndex--;renderQuiz();slideCard('prev');$('view-quiz').scrollTop=0;}
  }
}
function slideCard(dir){
  var qc=$('quizCard');if(!qc)return;
  qc.classList.add(dir==='next'?'quiz-slide-next':'quiz-slide-prev');
  setTimeout(function(){qc.classList.remove('quiz-slide-next','quiz-slide-prev');},360);
}
